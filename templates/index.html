Paste this script in your Supabase SQL Editor to enable remote table indexing:

CREATE TABLE IF NOT EXISTS public.edugenie_history (
    id BIGSERIAL PRIMARY KEY,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    module_type VARCHAR(50) NOT NULL,
    prompt_text TEXT NOT NULL,
    response_text TEXT,
    metadata JSONB DEFAULT '{}'::jsonb
);
CREATE TABLE IF NOT EXISTS public.quiz_scores (
    id BIGSERIAL PRIMARY KEY,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    topic TEXT NOT NULL,
    score INT NOT NULL,
    total_questions INT NOT NULL
);
ALTER TABLE public.edugenie_history ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.quiz_scores ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public read/write" ON public.edugenie_history FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public read/write scores" ON public.quiz_scores FOR ALL USING (true) WITH CHECK (true);
